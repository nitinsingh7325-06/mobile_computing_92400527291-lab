package commy.example.page;

import android.os.Bundle;
import android.view.View;
import android.widget.Button;
import android.widget.EditText;
import android.widget.Toast;

import androidx.activity.EdgeToEdge;
import androidx.appcompat.app.AppCompatActivity;

public class MainActivity extends AppCompatActivity implements Main {

    EditText text;
    Button submitbtn;
    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        EdgeToEdge.enable(this);
        setContentView(R.layout.activity_main);
        text = findViewById(R.id.text);
        submitbtn = findViewById(R.id.submitbtn);
        submitbtn.setOnClickListener((View.OnClickListener) this);

    }
    @Override
    public void onClick(View view){
        Toast.makeText(this, text.getText().toString(), Toast.LENGTH_SHORT).show();
    }
}

package com.cp.contentprovider;

import android.content.*;
import android.database.*;
import android.database.sqlite.*;
import android.net.Uri;

public class StudentProvider extends ContentProvider {

    public static final String AUTHORITY =
            "com.example.contentprovider.provider";

    public static final Uri URI =
            Uri.parse("content://" + AUTHORITY + "/students");

    SQLiteDatabase db;

    @Override
    public boolean onCreate() {
        SQLiteOpenHelper helper = new SQLiteOpenHelper(
                getContext(), "StudentDB", null, 1) {

            @Override
            public void onCreate(SQLiteDatabase db) {
                db.execSQL("CREATE TABLE students (" +
                        "id INTEGER PRIMARY KEY AUTOINCREMENT," +
                        "roll TEXT, name TEXT)");
            }

            @Override
            public void onUpgrade(SQLiteDatabase db,
                                  int oldVersion, int newVersion) {

                db.execSQL("DROP TABLE IF EXISTS students");

                onCreate(db);
            }
        };

        db = helper.getWritableDatabase();
        return true;
    }

    @Override
    public Cursor query(Uri uri, String[] p, String s,
                        String[] a, String order) {
        return db.query("students", p, s, a,
                null, null, order);
    }

    @Override
    public Uri insert(Uri uri, ContentValues values) {
        long id = db.insert("students", null, values);
        return Uri.withAppendedPath(URI, String.valueOf(id));
    }

    @Override
    public int update(Uri uri, ContentValues v,
                      String s, String[] a) {
        return db.update("students", v, s, a);
    }

    @Override
    public int delete(Uri uri, String s, String[] a) {
        return db.delete("students", s, a);
    }

    @Override
    public String getType(Uri uri) {
        return "vnd.android.cursor.dir/vnd.students";
    }
}
